function criarPedido() { var nomeServico = document.getElementById("nomeServico").value; var quantidade = document.getElementById("quantidade").value; var tamanhoP = document.getElementById("tamanhoP").value; var tamanhoM = document.getElementById("tamanhoM").value; var tamanhoG = document.getElementById("tamanhoG").value; var tamanhoGG = document.getElementById("tamanhoGG").value; var corMalha = document.getElementById("corMalha").value; var arte = document.getElementById("arte").files[0]; var dataEntrega = document.getElementById("dataEntrega").value; var observacoes = document.getElementById("observacoes").value; /* VERIFICA SE OS CAMPOS PRINCIPAIS FORAM PREENCHIDOS */ if ( nomeServico == "" || quantidade == "" || corMalha == "" || dataEntrega == "" ) { alert("Preencha os campos obrigatórios."); return; } /* FORMATA A DATA */ var partesData = dataEntrega.split("-"); var dataFormatada = partesData[2] + "/" + partesData[1] + "/" + partesData[0]; /* CRIA O PEDIDO */ var pedido = { nomeServico: nomeServico, quantidade: quantidade, tamanhoP: tamanhoP, tamanhoM: tamanhoM, tamanhoG: tamanhoG, tamanhoGG: tamanhoGG, corMalha: corMalha, dataEntrega: dataFormatada, observacoes: observacoes, imagem: "" }; /* SE EXISTIR UMA IMAGEM */ if (arte) { var leitor = new FileReader(); leitor.onload = function(event) { pedido.imagem = event.target.result; salvarPedido(pedido); }; leitor.readAsDataURL(arte); } else { salvarPedido(pedido); } } /* SALVAR PEDIDO */ function salvarPedido(pedido) { var pedidos = JSON.parse( localStorage.getItem("pedidos") ) || []; pedidos.push(pedido); localStorage.setItem( "pedidos", JSON.stringify(pedidos) ); /* MOSTRA A ESPECIFICAÇÃO */ var resultado = document.getElementById("resultado"); resultado.innerHTML = `
Nome do Serviço: ${pedido.nomeServico}
Quantidade: ${pedido.quantidade} peças
Tamanhos:

P: ${pedido.tamanhoP}

M: ${pedido.tamanhoM}

G: ${pedido.tamanhoG}

GG: ${pedido.tamanhoGG}

Cor da Malha: ${pedido.corMalha}
Arte do Serviço: ${ pedido.imagem ? `` : "Nenhuma imagem selecionada" }
Data de Entrega: ${pedido.dataEntrega}
Observações:

${pedido.observacoes}

`; /* VOLTA PARA O HALL */ setTimeout(function() { window.location.href = "hall.html"; }, 3000); } /* BOTÃO VOLTAR */ function voltarHall() { window.location.href = "hall.html"; } }
